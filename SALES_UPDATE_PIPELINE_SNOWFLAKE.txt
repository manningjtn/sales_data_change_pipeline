USE ROLE ACCOUNTADMIN;
USE WAREHOUSE COMPUTE_WH;
CREATE OR REPLACE DATABASE SALES_DATA_CHANGE;
CREATE OR REPLACE SCHEMA CHANGE_SCHEMA;

CREATE OR REPLACE FILE FORMAT SALES_DATA_CHANGE.CHANGE_SCHEMA.CSV_FORMAT
  TYPE = CSV
  SKIP_HEADER = 1
  FIELD_OPTIONALLY_ENCLOSED_BY = '"'
  NULL_IF = ('', 'NULL', 'null')
  EMPTY_FIELD_AS_NULL = TRUE;

CREATE OR REPLACE STORAGE INTEGRATION SALES_INT
TYPE = EXTERNAL_STAGE
STORAGE_PROVIDER = 'S3'
ENABLED = TRUE
STORAGE_AWS_ROLE_ARN = 'arn:aws:iam::655691758300:role/SNOWFLAKE-S3-SALES-DATA-ROLE'
STORAGE_ALLOWED_LOCATIONS = ('s3://sales-data-update-with-scdtype1/');

DESC STORAGE INTEGRATION SALES_INT;

CREATE OR REPLACE STAGE SALES_DATA_UPDATE
URL = 's3://sales-data-update-with-scdtype1/'
STORAGE_INTEGRATION = SALES_INT
FILE_FORMAT = CSV_FORMAT;

ls @SALES_DATA_UPDATE/full-data/;

CREATE OR REPLACE TABLE SALES_DATA_CHANGE.CHANGE_SCHEMA.FULL_DATA (
    CUSTOMERNAME VARCHAR,
    PHONE VARCHAR,
    ADDRESSLINE1 VARCHAR,
    ADDRESSLINE2 VARCHAR,
    CITY VARCHAR,
    STATE VARCHAR,
    POSTALCODE VARCHAR,
    COUNTRY VARCHAR,
    TERRITORY VARCHAR,
    CONTACTFIRSTNAME VARCHAR,
    CONTACTLASTNAME VARCHAR,
    UPDATE_DTS TIMESTAMP_NTZ,
    LOAD_DTS TIMESTAMP_NTZ
);

CREATE OR REPLACE PIPE SALES_DATA_CHANGE.CHANGE_SCHEMA.FULL_DATA_PIPE
AUTO_INGEST = TRUE
AWS_SNS_TOPIC = 'arn:aws:sns:us-east-1:655691758300:github-s3-snowflake-full-data'
AS
COPY INTO SALES_DATA_CHANGE.CHANGE_SCHEMA.FULL_DATA
FROM (
    SELECT
        $1 AS CUSTOMERNAME,
        $2 AS PHONE,
        $3 AS ADDRESSLINE1,
        $4 AS ADDRESSLINE2,
        $5 AS CITY,
        $6 AS STATE,
        $7 AS POSTALCODE,
        $8 AS COUNTRY,
        $9 AS TERRITORY,
        $10 AS CONTACTFIRSTNAME,
        $11 AS CONTACTLASTNAME,
        CURRENT_TIMESTAMP() AS UPDATE_DTS,
        CURRENT_TIMESTAMP() AS LOAD_DTS
    FROM @SALES_DATA_UPDATE/full-data/
)
FILE_FORMAT = (FORMAT_NAME = 'CSV_FORMAT')
ON_ERROR = 'CONTINUE';

SELECT *
FROM FULL_DATA
LIMIT 10;

CREATE OR REPLACE TABLE CHANGE_DATA_LANDING 
(
    CUSTOMERNAME VARCHAR,
    PHONE VARCHAR,
    ADDRESSLINE1 VARCHAR,
    ADDRESSLINE2 VARCHAR,
    CITY VARCHAR,
    STATE VARCHAR,
    POSTALCODE VARCHAR,
    COUNTRY VARCHAR,
    TERRITORY VARCHAR,
    CONTACTFIRSTNAME VARCHAR,
    CONTACTLASTNAME VARCHAR
);

CREATE OR REPLACE PIPE SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_PIPE
AUTO_INGEST = TRUE
AWS_SNS_TOPIC = 'arn:aws:sns:us-east-1:655691758300:github-s3-snowflake-change-data'
AS
COPY INTO SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_LANDING
FROM (
    SELECT
        $1 AS CUSTOMERNAME,
        $2 AS PHONE,
        $3 AS ADDRESSLINE1,
        $4 AS ADDRESSLINE2,
        $5 AS CITY,
        $6 AS STATE,
        $7 AS POSTALCODE,
        $8 AS COUNTRY,
        $9 AS TERRITORY,
        $10 AS CONTACTFIRSTNAME,
        $11 AS CONTACTLASTNAME
    FROM @SALES_DATA_UPDATE/change-data/
)
FILE_FORMAT = (FORMAT_NAME = 'CSV_FORMAT')
ON_ERROR = 'CONTINUE';

CREATE OR REPLACE STREAM SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_LANDING_STREAM
ON TABLE SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_LANDING
APPEND_ONLY = TRUE;

USE ROLE ACCOUNTADMIN;
USE WAREHOUSE COMPUTE_WH;
USE DATABASE SALES_DATA_CHANGE;
USE SCHEMA CHANGE_SCHEMA;

CREATE OR REPLACE PROCEDURE SALES_DATA_CHANGE.CHANGE_SCHEMA.UPDATE_FULL_DATA_SCD1()
RETURNS VARCHAR
LANGUAGE JAVASCRIPT
EXECUTE AS OWNER
AS
$$

try {

snowflake.execute({sqlText:`BEGIN TRANSACTION;`});

snowflake.execute({sqlText:`
CREATE OR REPLACE TEMPORARY TABLE SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_TEMP AS
SELECT
    CUSTOMERNAME,
    PHONE,
    ADDRESSLINE1,
    ADDRESSLINE2,
    CITY,
    STATE,
    POSTALCODE,
    COUNTRY,
    TERRITORY,
    CONTACTFIRSTNAME,
    CONTACTLASTNAME,
    CURRENT_TIMESTAMP(6) AS UPDATE_DTS
FROM SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_STREAM;
`});

snowflake.execute({sqlText:`
MERGE INTO SALES_DATA_CHANGE.CHANGE_SCHEMA.FULL_DATA TGT
USING SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_TEMP TMP
ON TGT.CUSTOMERNAME = TMP.CUSTOMERNAME

WHEN MATCHED THEN UPDATE SET
    TGT.PHONE = TMP.PHONE,
    TGT.ADDRESSLINE1 = TMP.ADDRESSLINE1,
    TGT.ADDRESSLINE2 = TMP.ADDRESSLINE2,
    TGT.CITY = TMP.CITY,
    TGT.STATE = TMP.STATE,
    TGT.POSTALCODE = TMP.POSTALCODE,
    TGT.COUNTRY = TMP.COUNTRY,
    TGT.TERRITORY = TMP.TERRITORY,
    TGT.CONTACTFIRSTNAME = TMP.CONTACTFIRSTNAME,
    TGT.CONTACTLASTNAME = TMP.CONTACTLASTNAME,
    TGT.UPDATE_DTS = TMP.UPDATE_DTS

WHEN NOT MATCHED THEN INSERT
(
    CUSTOMERNAME,
    PHONE,
    ADDRESSLINE1,
    ADDRESSLINE2,
    CITY,
    STATE,
    POSTALCODE,
    COUNTRY,
    TERRITORY,
    CONTACTFIRSTNAME,
    CONTACTLASTNAME,
    UPDATE_DTS,
    LOAD_DTS
)
VALUES
(
    TMP.CUSTOMERNAME,
    TMP.PHONE,
    TMP.ADDRESSLINE1,
    TMP.ADDRESSLINE2,
    TMP.CITY,
    TMP.STATE,
    TMP.POSTALCODE,
    TMP.COUNTRY,
    TMP.TERRITORY,
    TMP.CONTACTFIRSTNAME,
    TMP.CONTACTLASTNAME,
    TMP.UPDATE_DTS,
    CURRENT_TIMESTAMP(6)
);
`});

snowflake.execute({sqlText:`COMMIT;`});

return "SCD Type 1 Stored Procedure Executed Successfully";

}
catch (err)
{
    snowflake.execute({sqlText:`ROLLBACK;`});
    throw 'Error: ' + err;
}

$$;

CREATE OR REPLACE TASK SALES_DATA_CHANGE.CHANGE_SCHEMA.UPDATE_FULL_DATA_TASK
WAREHOUSE = COMPUTE_WH
SCHEDULE = '1 HOUR'
WHEN
SYSTEM$STREAM_HAS_DATA('SALES_DATA_CHANGE.CHANGE_SCHEMA.CHANGE_DATA_STREAM')
AS
CALL SALES_DATA_CHANGE.CHANGE_SCHEMA.UPDATE_FULL_DATA_SCD1();

ALTER TASK SALES_DATA_CHANGE.CHANGE_SCHEMA.UPDATE_FULL_DATA_TASK RESUME;
